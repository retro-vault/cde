foreach(tool dtsrcreate dtsrdbrec dtsrhan dtsrload dtsrindex dtsrdelete huffcode dtsrkdump dtsrclean)
    set(source ${tool}.c)
    if(tool STREQUAL "dtsrdelete")
        set(source tomita.c)
    endif()
    add_executable(${tool} ${source})
    target_include_directories(${tool} PRIVATE ${PROJECT_SOURCE_DIR}/lib/DtSearch ${PROJECT_SOURCE_DIR}/lib/DtSearch/raima)
    target_compile_definitions(${tool} PRIVATE MAIN_PROGRAM NO_XLIB)
    target_link_libraries(${tool} PRIVATE DtSearch DtSvc cde_build_options m)
    install(TARGETS ${tool} RUNTIME DESTINATION bin)
endforeach()
