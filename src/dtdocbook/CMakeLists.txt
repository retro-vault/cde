set(docbook_data ${CDE_DATA_TOP}/dtdocbook)
set(docbook_helpers ${CDE_LIBEXEC_DIR}/dtdocbook)
add_subdirectory(infolib)
add_subdirectory(instant)
add_executable(dtdocbook2infolib dtdocbook2infolib.c)
target_compile_definitions(dtdocbook2infolib PRIVATE DTDCBK_DATAROOTDIR="${docbook_data}" INFOLIB_LIBEXECDIR="${docbook_helpers}/infolib")
target_link_libraries(dtdocbook2infolib PRIVATE cde_build_options DtSvc tt)
install(TARGETS dtdocbook2infolib RUNTIME DESTINATION bin)
foreach(tool dtdocbook2sdl dtdocbook2man)
    configure_file(${tool}.in ${tool} @ONLY)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${tool} DESTINATION bin)
endforeach()
foreach(directory sgml spec tcl tpt locales dtsr)
    install(DIRECTORY ${directory}/ DESTINATION share/dt/dtdocbook/${directory})
endforeach()
add_executable(docbook_mkdbd ${PROJECT_SOURCE_DIR}/lib/DtSearch/tiny.c)
target_include_directories(docbook_mkdbd PRIVATE ${PROJECT_SOURCE_DIR}/lib/DtSearch/raima)
target_compile_definitions(docbook_mkdbd PRIVATE BASE_FILE_NAME="dtsearch")
target_link_libraries(docbook_mkdbd PRIVATE cde_build_options)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/dtsearch.dbd
    COMMAND ${CMAKE_COMMAND} -DTOOL=$<TARGET_FILE:docbook_mkdbd> -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR} -DLOG=${CMAKE_CURRENT_BINARY_DIR}/dtsearch.dbd -P ${PROJECT_SOURCE_DIR}/cmake/RunHelpTool.cmake
    DEPENDS docbook_mkdbd VERBATIM)
add_custom_target(docbook_database_schema ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/dtsearch.dbd)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/dtsearch.dbd DESTINATION share/dt/dtdocbook/dtsr)
