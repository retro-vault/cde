cmake_minimum_required(VERSION 3.22)
project(CDE VERSION 2.5.3 LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8
   OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "This port requires 64-bit x86 Linux")
endif()
file(REAL_PATH "${CMAKE_SOURCE_DIR}/build" CDE_BUILD_ROOT)
file(REAL_PATH "${CMAKE_BINARY_DIR}" CDE_BINARY_ROOT)
cmake_path(IS_PREFIX CDE_BUILD_ROOT "${CDE_BINARY_ROOT}" NORMALIZE CDE_BUILD_ALLOWED)
if(NOT CDE_BUILD_ALLOWED)
    message(FATAL_ERROR "Configure under build/, for example cmake -S . -B build/linux")
endif()
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "${PROJECT_SOURCE_DIR}/bin" CACHE PATH "Install staging root" FORCE)
endif()
set(CDE_INSTALLATION_TOP "${CMAKE_INSTALL_PREFIX}" CACHE PATH "Runtime installation prefix")
set(CDE_DATA_TOP "${CDE_INSTALLATION_TOP}/share/dt" CACHE PATH "Runtime desktop data directory")
set(CDE_CONFIGURATION_TOP "${CDE_INSTALLATION_TOP}/etc/dt" CACHE PATH "Runtime configuration directory")
set(CDE_LOGFILES_TOP "${CDE_INSTALLATION_TOP}/var/lib/dt" CACHE PATH "Runtime state directory")
set(CDE_CALENDAR_DIR "${CDE_LOGFILES_TOP}/calendar" CACHE PATH "Calendar service data directory")
set(CDE_LIBEXEC_DIR "${CDE_INSTALLATION_TOP}/libexec/cde" CACHE PATH "Runtime helper directory")
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INSTALL_LIBDIR lib)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
set(CDE_VERSION_MAJOR ${PROJECT_VERSION_MAJOR})
set(CDE_VERSION_MINOR ${PROJECT_VERSION_MINOR})
set(CDE_VERSION_MICRO ${PROJECT_VERSION_PATCH})
set(CDE_VERSION_DEV "")
configure_file(include/Dt/Dt.h.in include/Dt/Dt.h @ONLY)
configure_file(cmake/cde_config.h.in include/cde_config.h @ONLY)
include(CTest)
find_package(X11 REQUIRED COMPONENTS Xinerama Xrandr Xt Xext Xmu Xpm SM ICE)
include(cmake/CdeHeaders.cmake)
include(cmake/CdeDependencies.cmake)
include(cmake/CdeResources.cmake)
add_subdirectory(lib)
add_subdirectory(src)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

add_subdirectory(include)
install(FILES COPYING DESTINATION share/doc/cde)
